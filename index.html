// --- BRIDGE CONFIGURATION --- const BRIDGE_URL = "https://fire-ops-bridge.onrender.com/incidents"; const GOOGLE_URL = "https://script.google.com/macros/s/AKfycbz6CvzVCi8CKRrskgLl7IDZVpjl819vft0j5SRzY9AXWsnqmOEU-4TSwujBg5VlOneiyA/exec"; // --- PulsePoint Sync Logic --- async function syncPulsePoint() { const light = document.getElementById('bridge-status-light'); try { // Fetch from your verified live bridge const response = await fetch(BRIDGE_URL); if (!response.ok) throw new Error("Bridge response was not OK"); const incidents = await response.json(); // Update Status Light to Online (Green) light.classList.add('status-online'); const grid = document.getElementById('active-units-grid'); grid.innerHTML = ""; // Clear grid for fresh data if (incidents.length === 0) { grid.innerHTML = `
NO ACTIVE FIRE INCIDENTS REPORTED
`; return; } // Process each incident and its units incidents.forEach(inc => { if (inc.Unit && Array.isArray(inc.Unit)) { inc.Unit.forEach(unit => { // Match UnitID to your Google Sheet database const dbUnit = (window.unitMap || []).find(u => u.id === unit.UnitID); createUnitCard(unit, inc, dbUnit); }); } }); } catch (e) { console.error("Bridge Connection Lost:", e); light.classList.remove('status-online'); // Light turns gray } } // --- Card Creation Logic --- function createUnitCard(unitData, incidentData, dbData) { const grid = document.getElementById('active-units-grid'); const card = document.createElement('div'); card.className = 'unit-card'; // Fallback logic if unit isn't in your Google Sheet yet const unitName = dbData ? dbData.name : unitData.UnitID; const station = dbData ? dbData.station : "MUTUAL AID / UNKNOWN"; const logo = dbData ? dbData.logoUrl : ""; const bgStyle = logo ? `style="background-image: url('${logo}')"` : ""; card.innerHTML = `
${unitName}
${station}
${incidentData.CallSummary || 'FIRE CALL'}
${incidentData.FullAddress || 'ADDRESS PROTECTED'}
ID: ${unitData.UnitID} AGENCY: ${incidentData.agency_id || 'OR'}
`; grid.appendChild(card); } // --- Initialization --- window.onload = () => { // 1. Load Google Sheet Data (Station Directory & Logos) const s = document.createElement('script'); s.src = `${GOOGLE_URL}?callback=setup&ts=${Date.now()}`; document.body.appendChild(s); // 2. Start the PulsePoint bridge sync (every 20 seconds) setInterval(syncPulsePoint, 20000); }; // This function runs once Google Sheet data is loaded window.setup = function(raw) { let rows = Array.isArray(raw) ? raw : (raw.data || []); window.unitMap = rows.slice(1).map(row => ({ id: String(row[0] || "").toUpperCase().trim(), name: String(row[2] || ""), station: String(row[3] || "Unknown"), logoUrl: cleanGitHubUrl(row[8]) })); // Initial sync immediately after directory is ready syncPulsePoint(); }; function cleanGitHubUrl(url) { if (!url || typeof url !== 'string') return ""; return url.replace("github.com", "raw.githubusercontent.com").replace("/blob/", "/"); }
