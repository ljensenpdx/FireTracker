const app = express(); // AGGRESSIVE CORS FIX: Allows any browser to talk to this bridge app.use(cors({ origin: '*' })); let lastData = []; async function initBrowser() { const browser = await puppeteer.launch({ headless: false, executablePath: 'C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe', args: ['--window-size=1200,800'] }); const page = await browser.newPage(); await page.goto("https://web.pulsepoint.org/?agencies=00057", { waitUntil: 'networkidle2' }); setInterval(async () => { try { const data = await page.evaluate(() => { const results = []; const rows = Array.from(document.querySelectorAll('tr, div[role="row"]')); let reachedRecent = false; for (const row of rows) { if (row.innerText.includes('Recent')) { reachedRecent = true; break; } if (reachedRecent) continue; const lines = row.innerText.split('\n').map(l => l.trim()).filter(l => l.length > 0); const timeLine = lines.find(l => l.includes('AM') || l.includes('PM')); if (lines.length >= 2 && timeLine) { const statusBuckets = { "On Scene": new Set(), "To Hospital": new Set(), "En Route": new Set(), "Dispatched": new Set(), "Cleared": new Set() }; const unitElements = Array.from(row.querySelectorAll('div, span, td')).filter(el => /^[A-Z0-9?^*]{2,7}$/.test(el.innerText.trim())); unitElements.forEach(el => { const raw = el.innerText.trim(); const clean = raw.replace(/[?^*]/g, ''); const color = window.getComputedStyle(el).color; const [r, g, b] = color.match(/\d+/g).map(Number); if (r < 80 && g < 80 && b < 80) statusBuckets["Cleared"].add(clean); else if (r > 160 && g < 80) statusBuckets["On Scene"].add(clean); else if (r > 200 && g > 180) statusBuckets["To Hospital"].add(clean); else if (g > 160 && r < 120) statusBuckets["En Route"].add(clean); else statusBuckets["Dispatched"].add(clean); }); results.push({ agency: "CLACKAMAS FIRE", // <--- ADDED AGENCY type: lines[0], time: timeLine, address: lines.find(l => l.includes(' ') && l !== lines[0] && !l.includes('AM')) || "Location Private", unitStatuses: Object.entries(statusBuckets).filter(([_, u]) => u.size > 0).map(([s, u]) => `${s}: ${Array.from(u).join(', ')}`) }); } } return results; }); lastData = data; } catch (e) {} }, 5000); } app.get('/incidents', (req, res) => res.json(lastData)); app.listen(10000, () => { console.log("Bridge live on port 10000"); initBrowser();
